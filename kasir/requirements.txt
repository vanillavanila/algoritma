# Database Produk (Nested Dictionary)
data_produk = {
    "12345": {"nama": "Minyak", "harga": 40000, "stok": 10},
    "22345": {"nama": "Aqua", "harga": 3500, "stok": 11},
    "32345": {"nama": "Mie", "harga": 4000, "stok": 13},
    "42345": {"nama": "Sabun", "harga": 10000, "stok": 12},
    "52345": {"nama": "Sarden", "harga": 20000, "stok": 10}
}

# 1. Input Barcode
bacode = input("Masukan barcode / kode barang: ")

# 2. Pengecekan Keberadaan Barcode
if bacode in data_produk:
    produk = data_produk[bacode]
    print(f"Barang ditemukan: {produk['nama']} (Rp{produk['harga']})")
    print(f"Stok tersedia  : {produk['stok']}")

    # 3. Input Jumlah Beli
    qty = int(input("Masukan jumlah beli: "))

    # 4. Validasi Jumlah Beli dan Stok
    if qty > produk['stok']:
        print("\n❌ Gagal: Stok produk tidak cukup!")
    elif qty <= 0:
        print("\n❌ Gagal: Minimal beli 1!")
    else:
        # Perhitungan Total Belanja
        total_belanja = qty * produk['harga']
        print(f"Total Belanja  : Rp{total_belanja}")

        # 5. Input Nominal Pembayaran
        uang_diterima = int(input("Masukan jumlah pembayaran: Rp"))

        # 6. Validasi Uang Pembayaran
        if uang_diterima < total_belanja:
            print("\n❌ Gagal: Uang pembayaran tidak cukup!")
        else:
            # Perhitungan Kembalian & Pengurangan Stok
            kembalian = uang_diterima - total_belanja
            produk['stok'] -= qty   
            waktu_sekarang = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")

            # 7. Cetak Struk Transaksi
            print("\n" + "="*32)
            print("           X-MART")
            print("="*32)
            print(f"Tanggal : {waktu_sekarang}")
            print("-"*32)
            print(f"{produk['nama']}")
            print(f"  {qty} x {produk['harga']} = Rp{total_belanja}")
            print("-"*32)
            print(f"TOTAL     : Rp{total_belanja}")
            print(f"BAYAR     : Rp{uang_diterima}")
            print(f"KEMBALIAN : Rp{kembalian}")
            print("="*32)
            print("    TERIMA KASIH ATAS")
            print("      KUNJUNGAN ANDA")
            print("="*32)
            
else:
    print("❌ Barcode tidak ditemukan di database!")
